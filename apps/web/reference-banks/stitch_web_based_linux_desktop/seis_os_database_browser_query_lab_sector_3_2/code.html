WITH recent_anomalies AS (
  SELECT 
    e.event_id,
    e.timestamp,
    a.agent_name,
    e.payload->>'severity' AS severity_level
  FROM system_events e
  JOIN agent_registry a ON e.agent_ref = a.id
  WHERE e.timestamp > NOW() - INTERVAL '24 HOURS'
    AND e.payload->>'status' = 'CRITICAL'
)
SELECT * FROM recent_anomalies
ORDER BY timestamp DESC
LIMIT 100;
